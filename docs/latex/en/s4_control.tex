\section{Controlling the repo from the forward and dividend controls}\label{sec-control}

\begin{objectif}
(G2) Answer the question ``I have the forward control, I have the dividend control: how much is the repo control?'',
and measure, book by book, what Totem's split between repo and dividends changes in the control.
\end{objectif}

\subsection{Control lines}
Collect the repo rates at the pillars and the dividend levels per period in $\theta=(q, \ell)\in\R^N\times\R^m$. The
spot, the rate curve and the allocation profile are fixed, so the forwards at the pillars form a smooth function
$F(\theta)\in\R^N$. By \cref{prop-bootstrap}, for given forwards $F$ and levels $\ell$ there is a unique repo
$q=\psi(F, \ell)$ such that $F\bigl(\psi(F, \ell), \ell\bigr)=F$: this is the \emph{implied repo map}.

The FO marks are $\theta^{\fo}=(q^{\fo}, \ell^{\fo})$. Totem provides forwards $F^{\tot}$ and levels $\ell^{\tot}$,
hence $q^{\tot}=\psi(F^{\tot}, \ell^{\tot})$ and $\hat q=\psi(F^{\tot}, \ell^{\fo})$. A book has a value $V(\theta)$ and
FO sensitivities $s^q=\partial_qV(\theta^{\fo})\in\R^N$ and $s^\ell=\partial_\ell V(\theta^{\fo})\in\R^m$. Finally,
let
\[
\Delta\ell=\ell^{\tot}-\ell^{\fo}, \qquad\Delta q=q^{\tot}-q^{\fo}, \qquad\Delta\hat q=\hat q-q^{\fo}.
\]

\begin{definition}[Control lines]\label{def-lines}
\[
\text{dividend line}=\sprod{s^\ell}{\Delta\ell}, \qquad\text{repo line}=\sprod{s^q}{\Delta q}, \qquad
\text{forward line}=\sprod{s^q}{\Delta\hat q}.
\]
The \emph{total} is the sum of the dividend and repo lines. The forward line is the repo line one would get by keeping
the FO dividends: it depends on Totem only through its forwards.
\end{definition}

\begin{definition}[Forward-linear book]\label{def-fwdlinear}
A book is \emph{forward-linear} if $V(\theta)=\varphi\bigl(F(\theta)\bigr)$ for a $C^2$ function $\varphi$: its value
depends on the repo and the dividends only through the forwards. This is the case for European options valued with a
volatility surface that does not depend on the split, for forwards and for futures.
\end{definition}

\subsection{The Jacobian of the implied repo}
\begin{lemma}[Jacobian]\label{lem-jacobian}
The matrix $J=\partial_\ell\psi(F, \ell)\in\M_{N,m}(\R)$ is $J=-(\partial_qF)^{-1}\partial_\ell F$. The matrix
$\partial_qF$ is lower triangular with diagonal terms $\partial F_k/\partial q_k\le-T_kF_k<0$, hence invertible. To first
order,
\begin{equation}\label{eq-jacobian}
J_{kp}\approx-\frac{a_{kp}}{T_k\,(S_0-D_k)} .
\end{equation}
\end{lemma}
\begin{proof}
Differentiate $F\bigl(\psi(F, \ell), \ell\bigr)=F$ with respect to $\ell$: $\partial_qF\,J+\partial_\ell F=0$. With a
constant forward repo between pillars, $F_k$ depends only on $q_1, \dots, q_k$, hence the triangular form; the diagonal
term is $T_k\,\mathcal F_k'(q_kT_k)\le-T_kF_k$ by \cref{prop-bootstrap}. For the first order, hold the forwards fixed in
\eqref{eq-additivity}: $T_k\,\delta q_k=-\delta D_k/(S_0-D_k)=-a_{kp}\,\delta\ell_p/(S_0-D_k)$.
\end{proof}

\subsection{The control identity}
\begin{theorem}[Control identity]\label{thm-control}
For every book,
\begin{equation}\label{eq-control}
\text{dividend line}+\text{repo line}=\text{forward line}+\sprod{\eta}{\Delta\ell}+O\bigl(\norm{\Delta}^2\bigr),
\end{equation}
where $\eta=s^\ell+\tr{J}\,s^q\in\R^m$, $J$ is evaluated at $(F^{\tot}, \ell^{\fo})$ and
$\norm{\Delta}=\norm{\Delta\ell}+\norm{\Delta\hat q}$. We call $\sprod{\eta}{\Delta\ell}$ the \emph{non-forward line}
and $\eta$ the \emph{non-forward exposure}. If the book is forward-linear, $\eta=O(\norm{\Delta\hat q})$: the total
equals the forward line up to second order, and therefore does not depend on the Totem dividends to first order.
\end{theorem}
\begin{proof}
Expanding $\psi$ in $\ell$ gives $\psi(F^{\tot}, \ell^{\tot})=\psi(F^{\tot}, \ell^{\fo})+J\Delta\ell
+O(\norm{\Delta\ell}^2)$, that is, $\Delta q=\Delta\hat q+J\Delta\ell+O(\norm{\Delta\ell}^2)$. Hence
\[
\sprod{s^\ell}{\Delta\ell}+\sprod{s^q}{\Delta q}=\sprod{s^q}{\Delta\hat q}+\sprod{s^\ell+\tr{J}s^q}{\Delta\ell}
+O(\norm{\Delta}^2).
\]
If $V=\varphi\circ F$, then $\tr{s^q}=\nabla\varphi\,\partial_qF(\theta^{\fo})$ and
$\tr{s^\ell}=\nabla\varphi\,\partial_\ell F(\theta^{\fo})$, so
$\tr{\eta}=\nabla\varphi\,\bigl[\partial_\ell F+\partial_qF\,J\bigr](\theta^{\fo})$. At the point
$\theta'=(\hat q, \ell^{\fo})$, the bracket vanishes by \cref{lem-jacobian}; since $\theta'-\theta^{\fo}=(\Delta\hat q, 0)$
and the derivatives are Lipschitz, the bracket at $\theta^{\fo}$ is $O(\norm{\Delta\hat q})$, and
$\sprod{\eta}{\Delta\ell}=O(\norm{\Delta}^2)$.
\end{proof}

\begin{reponse}
Rewriting \eqref{eq-control}:
\[
\boxed{\;\text{repo line}=\text{forward line}-\text{dividend line}+\text{non-forward line}\;}
\]
up to second order. The repo control follows from the forward control and the dividend control. For a forward-linear
book, the non-forward line is zero: the repo control is exactly the difference of the other two, and the total
(dividend $+$ repo) does not depend on Totem's dividend contributors. The dividend panel enters only through the
non-forward exposure $\eta=s^\ell+\tr{J}s^q$, which is computed from the FO sensitivities and the bootstrap Jacobian
alone, with no revaluation.
\end{reponse}

The result even holds exactly under full revaluation.

\begin{theorem}[Exactness under full revaluation]\label{thm-exact}
If the book is forward-linear and all its maturities are Totem pillars, the full-revaluation control
$V(\theta^{\tot})-V(\theta^{\fo})=\varphi(F^{\tot})-\varphi\bigl(F(\theta^{\fo})\bigr)$, with
$\theta^{\tot}=(q^{\tot}, \ell^{\tot})$, does not depend on the Totem dividends $\ell^{\tot}$.
\end{theorem}
\begin{proof}
By construction of the implied repo, $F(\theta^{\tot})=F\bigl(\psi(F^{\tot}, \ell^{\tot}), \ell^{\tot}\bigr)=F^{\tot}$,
whatever $\ell^{\tot}$.
\end{proof}

\begin{remark}[Maturities between two pillars]
For a maturity between two pillars, the forward depends on the interpolated carry and on the dividends going ex in
between: a residual dependence on $\ell^{\tot}$ remains. It is of the order of the dividends going ex between the
maturity and the neighbouring pillars, that is, exactly the ex-date and profile effects of \cref{prop-signatures}, which
the detector watches. The second-order term of \eqref{eq-control} is written out in \cref{app-second}.
\end{remark}

\subsection{What the split costs a book}
\begin{proposition}[Materiality of the split]\label{prop-materiality}
Assume the consensus error on the dividend levels $\Delta\ell$ has covariance matrix $\Gamma^\ell$, the Totem forwards
being fixed. To first order, the variance of the total is $\tr{\eta}\,\Gamma^\ell\eta$, against
$\tr{(s^\ell)}\,\Gamma^\ell s^\ell$ for the dividend line alone. The share of the dividend line's uncertainty cancelled by
the repo line is therefore
\[
1-\sqrt{\frac{\tr{\eta}\,\Gamma^\ell\eta}{\tr{(s^\ell)}\,\Gamma^\ell s^\ell}} .
\]
\end{proposition}
\begin{proof}
By \cref{thm-control}, the total is the forward line, which does not depend on $\ell^{\tot}$, plus
$\sprod{\eta}{\Delta\ell}$.
\end{proof}

\begin{example}[Materiality]\label{ex-materiality}
With a consensus dispersion of \pct{5} of each period, ten contributors and a correlation $e^{-|c_p-c_{p'}|/2}$ between
periods with mid-points $c_p$ and $c_{p'}$:

\vMatA{} of the dividend line's uncertainty cancels for book A, \vMatB{} for B and \vMatC{} for C.
\end{example}

\subsection{When the split matters}
The non-forward exposure $\eta$ is not zero for dividend swaps and futures (pure dividend exposure), for stock borrow
fees and the funding legs of total return swaps (pure repo exposure), for American exercise (the early-exercise
boundary depends on dividends and repo separately) and for exotics valued with discrete dividend dynamics. For these
books, the split must be controlled directly, and the quality of the implied repo curve matters: this is the subject
of \cref{sec-detection}.

\begin{retenir}
\begin{itemize}[leftmargin=12pt]
\item Control the \emph{forward line} (Totem forwards, FO dividends) and the \emph{non-forward line}
$\sprod{\eta}{\Delta\ell}$; the latter needs the Totem dividends only where $\eta\ne0$.
\item Large dividend and repo lines of opposite signs, with a small total, are a split artefact, not a mismark.
\item In prudent valuation~\cite{eu2016}, aggregating uncertainties per parameter without offsetting counts a split
artefact twice; taking the forward as the parameter where $\eta\approx0$ removes the double count.
\end{itemize}
\end{retenir}
