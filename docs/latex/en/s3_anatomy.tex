\section{Anatomy of the implied repo}\label{sec-anatomy}

\subsection{The panel gap}
For a pillar $T$ write $\bA(T)$ and $\DA(T)$ for the averages of the repo and dividend views over the option panel,
and $\DB(T)$ for the dividend consensus of the dividend panel after allocation. Views are perturbations of a common
centre, $b_j=b_0+\delta b_j$ and $D_j=D_0+\delta D_j$, of order $\delta$.

\begin{proposition}[Panel-gap decomposition]\label{prop-gap}
Under \cref{ass-consistent,ass-panels,ass-consensus},
\begin{equation}\label{eq-gap}
b^\T(T)=\bA(T)+\frac{\DA(T)-\DB(T)}{T\,\bigl(S-D_0(T)\bigr)}+O(\delta^2).
\end{equation}
\end{proposition}
\begin{proof}
For $j\in\Aset$, \cref{lem-additivity} gives
$\ln F_j=\ln F_0-T\,\delta b_j-\delta D_j/(S-D_0)+O(\delta^2)$. The consensus forward is the mean of the $F_j$, and
$\ln$ of a mean equals the mean of the $\ln$ up to a variance term of order $\delta^2$, so
$\ln\overline{F}=\ln F_0-T\,\overline{\delta b}_{\Aset}-\overline{\delta D}_{\Aset}/(S-D_0)+O(\delta^2)$. The
controller allocates the dividend consensus, $\DB=D_0+\overline{\delta D}_{\Bset}$, and bootstraps:
$B^\T=R-\ln\overline{F}+\ln(S-\DB)
=B_0+T\,\overline{\delta b}_{\Aset}+\bigl(\overline{\delta D}_{\Aset}-\overline{\delta D}_{\Bset}\bigr)/(S-D_0)+O(\delta^2)$.
Divide by $T$ and use $b_0+\overline{\delta b}_{\Aset}=\bA$.
\end{proof}

The implied repo is the option panel's own repo plus a \emph{panel gap}: the difference between the option
contributors' dividend view, which is never observed, and the dividend panel's, divided by $T(S-D)$. For a single
contributor the forward and dividend errors cancel in the implied repo; across two panels they do not.

\begin{corollary}[Size of the gap]\label{cor-gap}
Let the dividend view errors at $T$ be i.i.d.\ with variance $\sigma_D^2(T)$ in both populations, and let the two
panels share $m$ contributors who use the same dividend view in both services. Then
\begin{equation}\label{eq-gap-sd}
\sd\bigl[b^\T-\bA\bigr]=\frac{\sigma_D(T)}{T\,(S-D(T))}\sqrt{\frac1{n_A}+\frac1{n_B}-\frac{2m}{n_An_B}}.
\end{equation}
\end{corollary}
\begin{proof}
$\DA-\DB=\sum_j(a_j-c_j)\,\delta D_j$ with $a_j=1/n_A$ on $\Aset$, $c_j=1/n_B$ on $\Bset$ and zero elsewhere. Its
variance is $\sigma_D^2\sum_j(a_j-c_j)^2=\sigma_D^2\bigl[m(1/n_A-1/n_B)^2+(n_A-m)/n_A^2+(n_B-m)/n_B^2\bigr]$, which
simplifies to $\sigma_D^2\,(1/n_A+1/n_B-2m/(n_An_B))$.
\end{proof}

With identical panels ($m=n_A=n_B$) the gap vanishes; with disjoint panels it is largest. The dispersion
$\sigma_D(T)$ of a cumulative dividend grows with $T$, roughly like $T$ or faster, so the gap is small in carry at the
short end but does not shrink in repo at the long end.

of the synthetic EU-stock category. The closed form \eqref{eq-gap-sd} with $m=0$ and a contributor-level
simulation with no repo disagreement give

\end{example}

\subsection{Contributor drop-off}
\begin{proposition}[Means of nested panels]\label{prop-nested}
Let contributor $j$'s error at tenor $k$ be $e_{jk}$, independent across contributors, with $\Var e_{jk}=\sigma_k^2$
and $\Cov(e_{jk},e_{jl})=\rho_{kl}\sigma_k\sigma_l$. If the panels are nested with sizes $n_k$, the panel means
$\bar e_k$ satisfy
\begin{equation}\label{eq-nested}
\Cov(\bar e_k,\bar e_l)=\frac{\rho_{kl}\,\sigma_k\sigma_l}{\max(n_k,n_l)}.
\end{equation}
In particular, with persistent views ($\rho=1$, $\sigma_k=\sigma_l=\sigma$) and $n_l\le n_k$,
$\sd(\bar e_l-\bar e_k)=\sigma\sqrt{1/n_l-1/n_k}$.
\end{proposition}
\begin{proof}
$\Cov(\bar e_k,\bar e_l)=\frac{1}{n_kn_l}\sum_{j\le\min(n_k,n_l)}\rho_{kl}\sigma_k\sigma_l
=\frac{\min(n_k,n_l)}{n_kn_l}\rho_{kl}\sigma_k\sigma_l$, and $\min/(n_kn_l)=1/\max$. Then
$\Var(\bar e_l-\bar e_k)=\sigma^2(1/n_l+1/n_k-2/n_k)=\sigma^2(1/n_l-1/n_k)$.
\end{proof}

\begin{example}
In units of the contributor dispersion, the consensus jumps by itself by $\vJumpA\,\sigma$ when a panel shrinks from
$16$ to $12$, $\vJumpB\,\sigma$ from $12$ to $7$, $\vJumpC\,\sigma$ from $15$ to $7$ and $\vJumpD\,\sigma$ from $10$
to $4$. A jump of that size at a panel drop is noise, not information; the noise model of \cref{sec-noise} reproduces
it through \cref{eq-nested}.
\end{example}

\subsection{Ex-date timing splits}
\begin{proposition}[Timing split]\label{prop-timing}
Let a dividend $d_i$ go ex at $t_i\in(T_k,T_{k+1})$ in the dividend allocation, and let a fraction $p$ of the option
panel place it before $T_k$ (all place it before $T_{k+1}$). Writing $\tilde d_i=d_i/G(t_i)$, to first order
\begin{enumerate}[label=(\roman*)]
\item the carry difference moves at $T_k$ only: $\delta y_k=p\,\tilde d_i/(S-D_k)$, i.e.\ a repo spike
$p\,\tilde d_i/\bigl(T_k(S-D_k)\bigr)$;
\item in forward repo between pillars this is a dipole: $+\delta y_k/(T_k-T_{k-1})$ then $-\delta y_k/(T_{k+1}-T_k)$;
\item the dispersion of contributor log-forwards at $T_k$ rises by $p(1-p)\,\tilde d_i^{\,2}/(S-D_k)^2$ in variance.
\end{enumerate}
\end{proposition}
\begin{proof}
(i) At $T_k$ a fraction $p$ of the option panel includes the dividend, so $\ln\overline{F}_k$ falls by
$p\,\tilde d_i/(S-D_k)$ (\cref{lem-additivity}) while the allocated $D^\T_k$ excludes it; by \cref{eq-additivity}
$B^\T_k$ rises by the same amount. Before $T_k$ nobody includes the dividend, after $T_{k+1}$ everybody does, so no
other pillar moves. (ii) Forward repo is the slope of $B$ between pillars. (iii) Law of total variance: the
contributors form two groups whose mean log-forwards differ by $\tilde d_i/(S-D_k)$, with weights $p$ and $1-p$.
\end{proof}

\begin{example}\label{ex-timing}
In the gallery example of \cref{fig-gallery}(c) a dividend of $2.18$ goes ex $17$ days after the six-month expiry and
$37\%$ of the option panel places it before. The spike is $0.37\times2.18/(0.5\times98)\approx1.6\%$, about $165$\,bp
of repo at one pillar. The contributor dispersion at that expiry is almost ten times that of its neighbours,
consistent with (iii).
\end{example}

\subsection{Stale ex-date profile}
\begin{proposition}[Profile shift]\label{prop-profile}
Let the option contributors use the true ex-dates $t_i$ while the allocation profile places them at $t_i+\Delta$.
Then, for $\Delta>0$ (symmetrically for $\Delta<0$),
\[
\delta y_k=\sum_{i:\,T_k\in[t_i,\,t_i+\Delta)}\frac{\tilde d_i}{S-D_k}:
\]
only pillars lying between a true ex-date and its shifted copy move.
\end{proposition}
\begin{proof}
$D^\T_k-D_k=\sum_i\tilde d_i\bigl(\one\{t_i+\Delta\le T_k\}-\one\{t_i\le T_k\}\bigr)$, which is $-\tilde d_i$ exactly
when $t_i\le T_k<t_i+\Delta$; apply \cref{eq-additivity}.
\end{proof}
For seasonal payers the affected pillars recur every year (for example the April expiries of European stocks), which
produces the zigzag that controllers recognise. Period ends are never affected, because the period totals are right.

\subsection{Dividend bucket errors}
\begin{proposition}[Bucket error]\label{prop-bucket}
If the Totem level of period $y$ is off by $\delta L$, then $\delta y_k=-A_{ky}\,\delta L/(S-D_k)$. For a period with
a single ex-date this is a step starting at the first pillar after the ex-date; in repo it is a jump that decays like
$1/T$.
\end{proposition}
\begin{proof}
\Cref{lem-allocation} gives $\delta D_k=A_{ky}\delta L$; apply \cref{eq-additivity} with the forwards unchanged.
$A_{ky}$ is zero before the ex-date and constant ($\approx1/G(t_i)$) after.
\end{proof}

\subsection{Spot snapshot and front dividends}
\begin{proposition}[Spot snapshot]\label{prop-spot}
If option contributors price off a spot $S(1+\varepsilon)$ while the controller uses $S$, then
\begin{equation}\label{eq-spot}
y_k=-\ln\Bigl(1+\frac{\varepsilon S}{S-D_k}\Bigr)\approx-\frac{\varepsilon S}{S-D_k}\qquad\text{for every }k,
\end{equation}
nearly constant across pillars (exactly constant without dividends), hence a repo error close to $-\varepsilon/T_k$. A
dividend before the first pillar mis-stated by $\delta d$ produces the same signature, $-\delta\tilde d/(S-D_k)$.
\end{proposition}
\begin{proof}
Each contributor's forward becomes $G(S(1+\varepsilon)-D)$, so $\ln\overline{F}_k$ moves by
$\ln\bigl((S(1+\varepsilon)-D_k)/(S-D_k)\bigr)$; by \cref{eq-additivity} $B^\T_k$ moves by the opposite. For the front
dividend, $D_k$ moves by $\delta\tilde d$ at every pillar.
\end{proof}

\subsection{Gross/net definitions and proportional biases}
\begin{proposition}[Proportional dividend bias]\label{prop-gross}
If every Totem dividend is $(1+\tau)$ times the dividend embedded in the forwards (for example gross against net of
withholding tax), then $y_k\approx-\tau D_k/(S-D_k)$. For a continuous dividend yield $q$, $D_k\approx SqT_k$ and
$y_k\approx-\tau q\,T_k$, which has exactly the shape of a repo level difference of $-\tau q$.
\end{proposition}
\begin{proof}
$\delta D_k=\tau D_k$ in \cref{eq-additivity}; for a yield, $D_k=S(1-e^{-qT_k})\approx SqT_k$ and $S-D_k\approx S$.
\end{proof}

\subsection{What the data cannot tell}
\begin{theorem}[Non-identifiability of the split]\label{thm-nonid}
Suppose European option prices depend on a contributor's views only through its forwards. For any contributor-level
transformation $(b_j,D_j)\mapsto(b_j+\delta_j,D'_j)$ with $\delta_j(T)\,T=\ln(S-D'_j(T))-\ln(S-D_j(T))$, every option
price, and hence every option-panel statistic (consensus, dispersion, count), is unchanged; the dividend-panel
statistics are unchanged too. The implied repo $b^\T$, a function of these statistics, is therefore unchanged, while
the option panel's repo $\bA$ moves by the average of the $\delta_j$. Consequently the panel gap $b^\T-\bA$ is not a
function of the observable data.
\end{theorem}
\begin{proof}
The transformation keeps $\ln F_j=R-B_j+\ln(S-D_j)$ fixed for every $j$ and every $T$, so each contributor's option
prices are unchanged and so is every statistic computed from them. The dividend panel is untouched. $b^\T$ is computed
from these statistics only, so it is unchanged, whereas $\bA$ shifts by the mean of the $\delta_j$, which can be any
value.
\end{proof}

\begin{corollary}
No estimator built from Totem consensus data alone (curve smoothing, clustering, neural networks) can remove the
panel gap. What remains possible is to model its size (\cref{sec-noise}), to recognise the structured mechanisms
by their signature (\cref{sec-detection}), and to avoid needing the split where the exposure does not
(\cref{sec-control}).
\end{corollary}

\begin{remark}[What breaks the degeneracy]
American options carry some information on the split through early exercise; overlapping contributors reduce the gap
(\cref{cor-gap}); external anchors pin either side directly: listed dividend futures, total-return futures that
quote a financing spread, securities-lending fees for the short end of single stocks.
\end{remark}
