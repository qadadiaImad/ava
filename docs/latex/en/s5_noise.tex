\section{A statistical model of the Totem--FO carry difference}\label{sec-noise}

\subsection{Observation model}
\begin{definition}[Observation model]\label{def-obs}
The detector observes the carry difference $y\in\R^K$ of \cref{eq-y} and models it as
\begin{equation}\label{eq-obs}
y=X\beta+\sum_{t\in\mathcal{S}}a_t\,t+\varepsilon,\qquad X=\bigl[\,T,\;T\ln T\,\bigr],\qquad\varepsilon\sim\N(0,\kappa\Sigma),
\end{equation}
where $X\beta$ is a smooth repo difference $\delta b(T)=\beta_0+\beta_1\ln T$ written in carry space (legitimate
disagreement, including a genuine FO mismark), $\mathcal{S}$ is a possibly empty set of anomaly templates
(\cref{sec-detection}), $\Sigma$ is the covariance of the consensus noise built below from Totem statistics, and
$\kappa>0$ is a single scale calibrated on clean history.
\end{definition}
$\beta_0$ is the repo difference at one year ($\ln1=0$); $\beta_1$ its slope in $\ln T$. Under the null hypothesis
$\mathcal{S}=\emptyset$.

\subsection{Covariance of consensus means}
By \cref{lem-additivity,prop-gap}, the consensus noise in carry is $\varepsilon_k=-e^F_k-e^D_k$, with $e^F_k$ the
error of the option-panel consensus $\ln\overline{F}_k$ and $e^D_k=\delta\DB(T_k)/(S-D_k)$ that of the allocated
dividend consensus. The two panels being independent (\cref{ass-panels} with $m=0$, the conservative case),
\begin{equation}\label{eq-sigma}
\Sigma=\Sigma^F+\Sigma^D.
\end{equation}
Both terms are covariances of means of nested panels, so \cref{prop-nested} applies: each is a contributor-level
covariance divided elementwise by $\max(n_k,n_l)$. What remains is to model the contributor-level covariance from the
dispersions Totem publishes.

\subsection{The option panel}
A contributor's log-forward error at $T$ is $-T\,\delta b_j-\delta D_j(T)/(S-D)$. Repo views are smooth (desks mark a
few pillars and interpolate), so the repo part is essentially a persistent level, with covariance
$\sigma_b^2T_kT_l$. The dividend part is cumulative: a persistent factor (a contributor who is bullish on dividends
is bullish on all years) plus increments that arrive with each dividend.

\begin{lemma}[Cumulative covariance]\label{lem-cumcov}
For a non-decreasing $s\in\R^K_{\ge0}$ and $w\in[0,1]$, the matrix
$C_{kl}=w\,s_ks_l+(1-w)\min(s_k,s_l)^2$ is positive semi-definite: it is the covariance of
$Z_k=\sqrt{w}\,s_k\,\xi+\sqrt{1-w}\,W(s_k^2)$, with $\xi\sim\N(0,1)$ independent of a standard Brownian motion $W$.
For $w=0$ the correlation between $k<l$ is $s_k/s_l$ (independent increments).
\end{lemma}
\begin{proof}
$\Cov(Z_k,Z_l)=w\,s_ks_l+(1-w)\Cov\bigl(W(s_k^2),W(s_l^2)\bigr)=w\,s_ks_l+(1-w)\min(s_k^2,s_l^2)$, and a covariance
matrix is positive semi-definite. With $w=0$, $\Cov(Z_k,Z_l)/\sqrt{\Var Z_k\Var Z_l}=s_k^2/(s_ks_l)=s_k/s_l$ for $k<l$.
\end{proof}

\begin{proposition}[Option-panel covariance]\label{prop-sigmaF}
Let $s_k=\sigma_{F,k}/F_k$ be the published dispersion of contributor forwards in log terms, and
$\sigma_b$ the dispersion of repo views. Splitting $s_k^2=\sigma_b^2T_k^2+s_k'^2$,
\begin{equation}\label{eq-sigmaF}
\Sigma^F_{kl}=\frac{\sigma_b^2\,T_kT_l+w\,s'_ks'_l+(1-w)\min(s'_k,s'_l)^2}{\max(n_{A,k},n_{A,l})}.
\end{equation}
\end{proposition}
\begin{proof}
Independence of repo and dividend views gives the sum of a rank-one term and the cumulative covariance of
\cref{lem-cumcov} for the dividend part; \cref{prop-nested} gives the division by the larger panel.
\end{proof}

\begin{remark}[Estimating $\sigma_b$]
At the shortest expiries dividend uncertainty is negligible, so $s_k\approx\sigma_bT_k$. The prototype takes the
median of $s_k/T_k$ over the first three pillars. The minimum would be biased low (the minimum of three noisy
estimates), which understates the uncertainty of the repo level and makes the FO-mismark test of
\cref{sec-detection} fire too often; this was observed on hard-to-borrow names and corrected.
\end{remark}

\subsection{The dividend panel}
\begin{proposition}[Dividend-panel covariance]\label{prop-sigmaD}
Let the period consensus errors have covariance
$(\Sigma_L)_{yz}=\rho_{yz}\,\sigma_{D,y}\sigma_{D,z}/\max(n_{B,y},n_{B,z})$, with $\rho_{yz}=e^{-|y-z|/\ell}$. Then, to
first order,
\begin{equation}\label{eq-sigmaD}
\Sigma^D=J_A\,\Sigma_L\,J_A^\top,\qquad (J_A)_{ky}=\frac{A_{ky}}{S-D_k}.
\end{equation}
\end{proposition}
\begin{proof}
$e^D_k=\delta D_k/(S-D_k)=\sum_yA_{ky}\delta L_y/(S-D_k)$ by \cref{lem-allocation}: a linear map of the period errors.
\end{proof}

\subsection{Dispersion spikes are symptoms, not noise}
\begin{proposition}[Masking]\label{prop-masking}
In the timing split of \cref{prop-timing}, let $s_k$ be the dispersion without the split and $\Delta=\tilde d_i/(S-D_k)$.
The test statistic for the shift at $T_k$, computed with the published (inflated) dispersion, is
\begin{equation}\label{eq-masking}
z=\frac{p\,\Delta}{\sqrt{s_k^2+p(1-p)\Delta^2}/\sqrt{n_{A,k}}},
\end{equation}
which stays bounded by $\sqrt{n_{A,k}\,p/(1-p)}$ however large the dividend, whereas with the clean dispersion $z$
grows linearly in $\Delta$.
\end{proposition}
\begin{proof}
\Cref{prop-timing}(i) gives the mean shift and (iii) the inflated variance. For $\Delta\to\infty$ the ratio tends to
$\sqrt{n_{A,k}}\,p/\sqrt{p(1-p)}$.
\end{proof}

\begin{example}
For the gallery example ($p=0.37$, $n_A=16$) the bound is $\sqrt{16\times0.37/0.63}\approx3.1$: just around the
family-wise critical value, so the split would often go unnoticed. The prototype therefore caps an isolated dispersion
spike at $1.5$ times the larger of its two neighbours before building $\Sigma$, and reports the ratio, which is itself
evidence that contributors disagree, usually about an ex-date (here $9.7$).
\end{example}

\begin{insight}
$\Sigma$ uses only what Totem publishes (dispersions and contributor counts) plus three structural settings: the
factor weight $w$, the correlation length $\ell$ of dividend periods and the repo dispersion read off the short end.
The single scale $\kappa$ absorbs a common bias of these settings; their \emph{shape} is what makes the detector
powerful, because it tells it which deviations are cheap (smooth, long-dated) and which are expensive (short-dated,
local).
\end{insight}
